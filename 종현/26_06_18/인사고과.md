import java.util.*;
class Solution {
    public int solution(int[][] scores) {      
        int aMax = scores[0][0];
        int bMax = scores[0][1];
        int critA = aMax;
        int critB = bMax;
        
        for(int[] score : scores) {
            int aCur = score[0];
            int bCur = score[1];
            if(aMax <= aCur && bMax <= bCur) {
                aMax = aCur;
                bMax = bCur;
            }
        }
        
        //정렬 기준
        Arrays.sort(scores, (a, b) -> Integer.compare(b[0]+b[1], a[0]+a[1]));
        
        int rank=1;
        int exception = 0;
        if(critA >= aMax || critB >= bMax) {
            for(int i=0; i<scores.length; i++) {
                if(scores[i][0]==critA && scores[i][1]==critB) return rank;
                /*
                int[][] scores = {
                    {3, 3},  // 완호
                    {6, 3},
                    {5, 2}
                }; 에서 2를 리턴해야하나 elseif로 비교하지 않으면 3이 리턴됨 
                */
                else if(scores[i][0] < aMax && scores[i][1] < bMax) {
                    exception++;
                    continue;
                }
                rank++;
            }
        }
        
        
        return -1;
    }
}
